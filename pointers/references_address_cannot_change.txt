#include <iostream>
//refences address cannot be changed , even if you assign new value
int main() {

    int x = 10;
    int& r = x;
    int y = 20;
    
    std::cout<<r<<std::endl;
    std::cout<<&r<<std::endl;
    std::cout<<&x<<std::endl;
    
    r = y;
    std::cout<<r<<std::endl;
     std::cout<<&r<<std::endl;
      std::cout<<&x<<std::endl;
       std::cout<<x<<std::endl;
    return 0;
}
