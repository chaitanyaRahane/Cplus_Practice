#include<iostream>

using namespace std;

struct  vector
{
	int* data = nullptr;
	int size = 0;

	vector() {};

	vector(int s) {
		size = s;
		data = new int[size] {};
	}

	//copy constructor
	vector(const vector& other) {
		size = other.size;
		data = new int[size];
		std::copy(other.data, other.data + size, data);
	}

	//move constrcutor
	vector(vector&& other) noexcept {
		data = other.data;
		size = other.size;
		other.data = nullptr;
		other.size = 0;
	}

	//move operator (&& = rvalue reference)
	// avoiding deep copies. good for performance
	vector& operator= (vector&& other) noexcept {
		if (this != &other) { // this check if same object is getting moved
			data = other.data;
			size = other.size;
			other.data = nullptr;
			other.size = 0;
			cout << "move operator" << endl;
		}
		return *this; // this allows asignement chaining
	}

	int& operator[](int index) {
		return data[index];
	}

	~vector() {
		cout << "called" << endl;
		delete[] data;
	}
};
