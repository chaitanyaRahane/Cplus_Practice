#include<iostream>
#include<thread>

using namespace std;

//multithreading is used to execute multiple functions of the same program at same time/ Simultaneously.
void f1(void) {
	for (int i = 0; i < 5; i++) {
		cout << "+";
	}
}
void f2(void) {
	for (int i = 0; i < 5; i++) {
		cout << "-";
	}
}
int main(void) {
	thread t1(f1);
	thread t2(f2);

	t1.join(); // Wait for t1 to finish
	t2.join(); // Wait for t2 to finish

	return 0;
}
